# Cups, glasses and menu icons for the drinks.
# See README.md in this folder for how these files work.
#
# Pixels marked @drink are the inside of the cup. The game fills them with
# the drink's own colors (from content/drinks.json) and empties them from the
# top as you sip; the color written here is what an empty cup shows there.

palette porcelain
  o #8a8174  outline
  W #fbf8f2  rim
  w #ebe6dc  porcelain
  v #c9c2b5  porcelain, shadow side
  h #ddd6ca  handle
  s #f3efe8  saucer
  S #c8c1b4  saucer edge
  d #d6cfc2 @drink  inside of the cup

palette glass
  O #9aa3a0  glass edge
  g #e3e8e4  thick glass
  G #f6f8f5  glass, catching the light
  d #dfe3dd @drink  inside of the glass
